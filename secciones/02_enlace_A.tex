\begin{frame}{Zona de Fresnel y obstáculo crítico}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \small
    Radio máximo en el centro del vano ($d_1=d_2=\EAdUno$ km, $f=\EAfrecNum$ GHz):
    \[
      R_1 = 17.32\sqrt{\frac{\EAdUno\times\EAdDos}{\EAfrecNum\times 8.72}} = \textbf{\EArFresnel}
    \]
    \begin{itemize}
      \item Despeje mínimo (60\,\%): \textbf{\EArSesenta}
      \item Obstáculo más cercano a la trayectoria: \textbf{edificio a 5.35 km} del origen.
      \item Con \EAunoAlt\ / \EAdosAlt\ se cumple despeje \EAdespejePct\
            $\Rightarrow$ sin pérdidas por difracción.
    \end{itemize}
    \column{0.45\textwidth}\centering
    \figura{A_obstaculo.jpg}{\linewidth}\\
    {\scriptsize Obstrucción más importante del enlace}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Determinación de las alturas mínimas de antena}
  \begin{itemize}
    \item Proceso iterativo: se evaluó bajar de \EAunoAlt\ (FCEFyN) y \EAdosAlt\ (Canal 10).
    \item Al reducir alturas, el elipsoide de Fresnel desciende y se acerca peligrosamente a la
          edificación.
    \item Sin datos LIDAR, UISP ignora edificios: acercar el haz al suelo implica chocar con
          estructuras no modeladas.
  \end{itemize}
  \begin{alertblock}{Conclusión}
    \EAunoAlt\ / \EAdosAlt\ son el \textbf{mínimo indispensable} para mantener el 60\,\% de
    despeje; subir más solo agrega costo y carga al viento.
  \end{alertblock}
\end{frame}

\begin{frame}{Atenuación de espacio libre y presupuesto de RF}
  \begin{columns}[T]
    \column{0.47\textwidth}
    \textbf{FSPL} ($d$ en km, $f$ en MHz)
    \begin{align*}
      \text{FSPL} &= 32.44 + 20\log(8.72) + 20\log(5500)\\
                  &\approx \textbf{126.06 dB}
    \end{align*}
    \small Frecuencia representativa de la banda: 5.5 GHz. UISP no expone el FSPL, solo la
    señal esperada.
    \column{0.5\textwidth}
    \small
    \begin{tabular}{@{}lr@{}}
      \toprule
      $P_{TX}$ (ajuste automático) & \EAptx \\
      $L_{TX}$, $L_{RX}$ (InnerFeed) & 0 dB \\

\begin{frame}{PIRE y verificación regulatoria}
  \[
    \text{PIRE} = P_{TX} - L_{TX} + G_{TX} = 8 - 0 + 23 = \textbf{31 dBm ($\approx$1.26 W)}
  \]
  \begin{itemize}
    \item ENACOM fija límites de PIRE en 5 GHz de \textbf{30 dBm (1 W) a 36 dBm (4 W)}
          según la sub-banda (5.15--5.85 GHz).
    \item Con 31 dBm el enlace se declara dentro de la normativa, con emisión controlada.
  \end{itemize}
  % REVISAR: 31 dBm supera 30 dBm; solo cumple si la sub-banda usada admite más de 30 dBm.
  % Definir la sub-banda (p. ej. 5725-5850 MHz) y citar la Res. 581-MM/2018.
\end{frame}

\begin{frame}{Sensibilidad, modulación y margen de enlace}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small
    $M = P_{RX} - S_{RX}$, con $P_{RX}=\text{\EAprxSim}$ y canal de 20 MHz\\[4pt]
    \begin{tabular}{@{}llcc@{}}
      \toprule
      \textbf{MCS} & \textbf{Mod.} & $S_{RX}$ & $M$ \\ \midrule
      MCS 0 & BPSK   & $-96$ dBm & 24 dB \\
      MCS 4 & 16QAM  & $-86$ dBm & \textcolor{verde}{\textbf{14 dB}} \\
      MCS 6 & 64QAM  & $-77$ dBm & 5 dB \\
      MCS 8 & 256QAM & $-65$ dBm & \textcolor{red}{$-7$ dB} \\
      \bottomrule
    \end{tabular}
    \column{0.45\textwidth}
    \small
    \begin{itemize}
      \item Más orden de modulación $\Rightarrow$ más señal requerida $\Rightarrow$ menos margen.
      \item BPSK/16QAM: margen $>$10 dB, muy estable.
      \item 256QAM: margen negativo, imposible de sostener; se negocia 16QAM o 64QAM.
      \item \textbf{Modulación adaptativa:} ante lluvia, desvanecimientos o más ruido, baja el
            MCS (BPSK/QPSK) en lugar de cortar el enlace.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Nivel de ruido y SNR}
  Ruido térmico para 20 MHz: \EAruido
  \[
    \text{SNR} = P_{RX} - P_{ruido} = -72 - (-101) = \textbf{\EAsnr}
  \]
  \begin{itemize}
    \item Adecuado para 16QAM y 64QAM; 256QAM suele exigir $>$32 dB.
    \item Garantiza baja BER en las modulaciones que el equipo negocia.
  \end{itemize}

\begin{frame}{Pérdidas adicionales, interferencias y extremo crítico}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Pérdidas por obstáculos}
      \small Alturas elegidas para salvar el edificio a 5.35 km con despeje \EAdespejePct\
      $\Rightarrow L_{adicionales}=0$ dB.
    \end{block}
    \begin{block}{Interferencias}
      \small Se propone \textbf{airView} para identificar señales, canales ocupados y ruido.
      El escaneo real queda para el relevamiento en sitio (\emph{site survey}); en banda libre
      no se puede garantizar ausencia de interferencias.
    \end{block}
    \column{0.48\textwidth}
    \begin{alertblock}{Extremo más crítico: FCEFyN (\EAunoAlt)}
      \small
      \begin{itemize}
        \item Zona de altísima densidad edilicia: routers y enlaces WISP elevan el ruido de fondo.
        \item Menor altura (\EAunoAlt\ vs.\ \EAdosAlt): más expuesta a fuentes en terrazas.
        \item Azimut \EAazUnoDos: el lóbulo de 23 dBi apunta al centro de Córdoba.
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Consideraciones regulatorias (ENACOM)}
  \small
  \begin{tabularx}{\textwidth}{@{}lL@{}}
    \toprule
    Banda & \EAfrec \\
    Condición de uso & Uso libre / compartido, no licenciado \\
    PIRE máxima & 30 dBm (1 W) a 36 dBm (4 W) según sub-banda; calculada: \EApire \\
    Restricciones & No causar interferencia perjudicial a servicios primarios y aceptar la recibida \\
    Uso compartido & Sin protección regulatoria frente a interferencias de terceros \\
    \bottomrule
  \end{tabularx}
  \vspace{6pt}
  \begin{alertblock}{Control de potencia en bandas compartidas}
    Transmitir con más potencia de la necesaria eleva el piso de ruido local y aumenta las
    colisiones, degradando el SNR de todos los enlaces de la zona, incluido el propio.
    Ajustar la PIRE es parte del diseño.
  \end{alertblock}
\end{frame}

      \item \textbf{Ajuste fino:} se fija un extremo y se microajusta el opuesto monitoreando
            RSSI y SNR (app móvil o LEDs).
      \item Se alterna entre estaciones hasta alcanzar $\approx$\EAprxSim\ y se ajustan las tuercas.
    \end{enumerate}
  \end{columns}
